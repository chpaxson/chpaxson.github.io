# Problem
A regular hexagon with sides of length 6 has an isosceles triangle attached to each side. Each of these triangles has two 
sides of length 8. The isosceles triangles are folded to make a pyramid with the hexagon as the base of the pyramid. 
What is the volume of the pyramid?

answers = $18$, $162$, $36\sqrt{21}$, $18\sqrt{138}$, $54\sqrt{21}$

# Solution

**Step 1 — Slant height of each triangular face.**
Each isosceles triangle has base 6 and two legs of 8. Its altitude (from the midpoint of the base to the apex) is:
$$m = \sqrt{8^2 - 3^2} = \sqrt{55}$$

When folded up, $m$ becomes the **slant height** of the pyramid — the distance from the midpoint of a base edge to the apex.

**Step 2 — Height of the pyramid.**
The **inradius** of a regular hexagon with side $s=6$ is $r = s\frac{\sqrt{3}}{2} = 3\sqrt{3}$.

By the Pythagorean theorem in the right triangle (inradius, height, slant height):
$$H = \sqrt{m^2 - r^2} = \sqrt{55 - 27} = \sqrt{28} = 2\sqrt{7}$$

**Step 3 — Volume.**
The area of a regular hexagon with side 6 is:
$$A = \frac{3\sqrt{3}}{2}s^2 = \frac{3\sqrt{3}}{2}(36) = 54\sqrt{3}$$

$$V = \frac{1}{3}AH = \frac{1}{3}(54\sqrt{3})(2\sqrt{7}) = \boxed{36\sqrt{21}}$$

The answer is $\textbf{(C)}$.

<script type="module">
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

const S = 6, LEG = 8, N = 6;
const inR = S * Math.sqrt(3) / 2;  // 3√3
const slant = Math.sqrt(LEG*LEG - (S/2)*(S/2)); // √55
const H = Math.sqrt(slant*slant - inR*inR);       // 2√7

// Hexagon vertices (flat, in XZ plane)
function hexVert(i) {
  const a = (Math.PI/3)*i;
  return [S*Math.cos(a), 0, S*Math.sin(a)];
}

// For each face i: base edge goes from hexVert[i] to hexVert[(i+1)%6]
// Mid of base edge = inR away from center; triangle apex = beyond that by slant
// Flat: triangle apex lies in XZ plane at distance inR + slant from center (along face normal)
// Folded: apex rises to (0, H, 0)

const FACE_COLORS = [0x3d8ef0, 0x9b6bff, 0x22cc66, 0xff6644, 0xffaa33, 0xff4488];

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0d0d1a);
const camera = new THREE.PerspectiveCamera(42, innerWidth/innerHeight, 0.01, 200);
camera.position.set(14, 12, 18);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
document.body.prepend(renderer.domElement);

const orbit = new OrbitControls(camera, renderer.domElement);
orbit.enableDamping = true; orbit.dampingFactor = 0.07;
orbit.target.set(0, H/2, 0); orbit.update();

scene.add(new THREE.AmbientLight(0x8899cc, 0.55));
const sun = new THREE.DirectionalLight(0xffffff, 1.4);
sun.position.set(10, 18, 12); scene.add(sun);
const fill = new THREE.DirectionalLight(0x4466ff, 0.35);
fill.position.set(-8, 4, -8); scene.add(fill);

// Pre-compute flat and folded apex positions for each face
const faceData = [];
for (let i = 0; i < N; i++) {
  const v0 = hexVert(i);
  const v1 = hexVert((i+1)%N);
  // midpoint of base edge
  const mx = (v0[0]+v1[0])/2, mz = (v0[2]+v1[2])/2;
  // outward normal direction (unit vec from center to mid)
  const len = Math.sqrt(mx*mx+mz*mz);
  const nx = mx/len, nz = mz/len;
  // flat apex: out in XZ plane by slant from mid
  const flatApex = [mx + nx*slant, 0, mz + nz*slant];
  // folded apex: (0, H, 0) for all faces
  const foldApex = [0, H, 0];
  faceData.push({ v0, v1, flatApex, foldApex });
}

// Build face meshes
const faceMeshes = faceData.map(({v0,v1,flatApex}, i) => {
  const geo = new THREE.BufferGeometry();
  const pos = new Float32Array(9);
  geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
  geo.setIndex([0,1,2]);
  geo.computeVertexNormals();
  const mesh = new THREE.Mesh(geo, new THREE.MeshStandardMaterial({
    color: FACE_COLORS[i], side: THREE.DoubleSide,
    transparent: true, opacity: 0.72, roughness: 0.35, depthWrite: false
  }));
  scene.add(mesh);
  return mesh;
});

// Hexagon base (always flat)
const baseShape = new THREE.Shape();
for (let i=0;i<N;i++) {
  const [x,,z] = hexVert(i);
  i===0 ? baseShape.moveTo(x,z) : baseShape.lineTo(x,z);
}
baseShape.closePath();
const baseGeo = new THREE.ShapeGeometry(baseShape);
// Rotate shape (it's in XY) to XZ plane
baseGeo.rotateX(-Math.PI/2);
const baseMesh = new THREE.Mesh(baseGeo, new THREE.MeshStandardMaterial({
  color: 0x334466, side: THREE.DoubleSide, transparent: true, opacity: 0.35
}));
scene.add(baseMesh);

// Edge lines
const edgeLines = [];
function makeEdgeLine(col) {
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(6), 3));
  const line = new THREE.Line(geo, new THREE.LineBasicMaterial({ color: col }));
  scene.add(line);
  return line;
}
const apexLines = faceData.map(() => makeEdgeLine(0xffffff));   // apex → v0
const apexLines2 = faceData.map(() => makeEdgeLine(0xffffff));  // apex → v1

// Height line (dashed-look: just two points)

function lerp3(a, b, t) {
  return [a[0]+t*(b[0]-a[0]), a[1]+t*(b[1]-a[1]), a[2]+t*(b[2]-a[2])];
}

function setLine(line, p1, p2) {
  const arr = line.geometry.attributes.position.array;
  arr[0]=p1[0]; arr[1]=p1[1]; arr[2]=p1[2];
  arr[3]=p2[0]; arr[4]=p2[1]; arr[5]=p2[2];
  line.geometry.attributes.position.needsUpdate = true;
}

function update(t) {
  const apexArr = faceData.map(({flatApex, foldApex}) => lerp3(flatApex, foldApex, t));

  faceData.forEach(({v0,v1}, i) => {
    const ap = apexArr[i];
    const pos = faceMeshes[i].geometry.attributes.position.array;
    pos[0]=v0[0]; pos[1]=v0[1]; pos[2]=v0[2];
    pos[3]=v1[0]; pos[4]=v1[1]; pos[5]=v1[2];
    pos[6]=ap[0]; pos[7]=ap[1]; pos[8]=ap[2];
    faceMeshes[i].geometry.attributes.position.needsUpdate = true;
    faceMeshes[i].geometry.computeVertexNormals();

    setLine(apexLines[i],  [v0[0],v0[1],v0[2]], ap);
    setLine(apexLines2[i], [v1[0],v1[1],v1[2]], ap);
  });

  // Height and inradius indicators (shown when mostly folded)

  heightLine.material.opacity = alpha;
  inRadLine.material.opacity  = alpha;
  heightLine.material.transparent = true;
  inRadLine.material.transparent  = true;

  // Use face 0 for inradius viz
  const [mx,,mz] = [faceData[0].v0[0]/2 + faceData[0].v1[0]/2, 0,
                    faceData[0].v0[2]/2 + faceData[0].v1[2]/2];
  setLine(heightLine, [0,0,0], [0,H*t,0]);
  setLine(inRadLine,  [0,0,0], [mx, 0, mz]);

  // Apex dot (average of all face apexes — converges to single point)
  const avgApex = apexArr.reduce((s,a)=>[s[0]+a[0],s[1]+a[1],s[2]+a[2]], [0,0,0])
    .map(v=>v/N);
  apexDot.position.set(...avgApex);
}
