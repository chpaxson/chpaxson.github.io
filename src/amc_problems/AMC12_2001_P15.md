# Problem
An insect lives on the surface of a regular tetrahedron with edges of length 1. It wishes to travel on the surface of the 
tetrahedron from the midpoint of one edge to the midpoint of the opposite edge. What is the length of the shortest 
such trip? (Note: Two edges of a tetrahedron are opposite if they have no common endpoint.)

answers = $\frac{1}{2}$, $\frac{\sqrt{2}}{2}$, $\frac{3}{4}$, $1$, $\frac{3}{2}$

# Solution

**Key idea — unfold the surface flat.**

Label the vertices $A, B, C, D$. The two opposite edges are $AB$ and $CD$; call their midpoints $M_1$ and $M_2$. Any surface path from $M_1$ to $M_2$ must cross at least two faces (since opposite edges share no face). The shortest two-face path lies in one of four equivalent unfoldings:

**Unfolding faces $ABC$ and $ACD$ across their shared edge $AC$:**

Place $ABC$ flat with $A=(0,0)$, $B=(1,0)$, $C=\!\left(\tfrac{1}{2},\tfrac{\sqrt3}{2}\right)$.

Rotating face $ACD$ about $AC$ into the plane sends $D$ to $D' = \!\left(-\tfrac{1}{2}, \tfrac{\sqrt3}{2}\right)$, forming a rhombus $ABD'C$ (all sides $= 1$).

$$M_1 = \text{mid}(AB) = \left(\tfrac{1}{2},0\right), \qquad M_2 = \text{mid}(CD') = \left(0,\tfrac{\sqrt3}{2}\right)$$

The straight-line distance in the flat unfolding is:

$$|M_1 M_2| = \sqrt{\left(\tfrac{1}{2}\right)^2 + \left(\tfrac{\sqrt3}{2}\right)^2} = \sqrt{\tfrac{1}{4}+\tfrac{3}{4}} = \boxed{1}$$

This straight line crosses $AC$ at its midpoint, confirming it stays within the two faces. By symmetry, all four two-face unfoldings give the same length. Any path through three or more faces (or via a vertex) is strictly longer.

The answer is $\textbf{(D)}\ 1$.

<script type="module">
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

// 3D coordinates of regular tetrahedron, edge=1
const s3 = Math.sqrt(3), s6 = Math.sqrt(6);
const A3 = new THREE.Vector3(0, 0, 0);
const B3 = new THREE.Vector3(1, 0, 0);
const C3 = new THREE.Vector3(0.5, s3/2, 0);
const D3 = new THREE.Vector3(0.5, s3/6, s6/3);

// Unfolded coordinates: rhombus in XY plane (z=0)
// ABC flat: A=(0,0,0), B=(1,0,0), C=(0.5, √3/2, 0)
// ACD unfolded across AC → D'=(-0.5, √3/2, 0)
const A2 = new THREE.Vector3(0,       0,       0);
const B2 = new THREE.Vector3(1,       0,       0);
const C2 = new THREE.Vector3(0.5,   s3/2,     0);
const Dp2= new THREE.Vector3(-0.5,  s3/2,     0); // D' unfolded

// Lerp between 3D and unfolded positions
function lerp(a3, a2, t) {
  return new THREE.Vector3(
    a3.x + t*(a2.x - a3.x),
    a3.y + t*(a2.y - a3.y),
    a3.z + t*(a2.z - a3.z)
  );
}

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0d0d1a);
const camera = new THREE.PerspectiveCamera(42, innerWidth/innerHeight, 0.01, 100);
camera.position.set(0.5, 1.2, 2.5);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
document.body.prepend(renderer.domElement);

const orbit = new OrbitControls(camera, renderer.domElement);
orbit.enableDamping = true; orbit.dampingFactor = 0.07;
orbit.target.set(0.5, 0.4, 0); orbit.update();

scene.add(new THREE.AmbientLight(0x8899cc, 0.6));
const sun = new THREE.DirectionalLight(0xffffff, 1.3);
sun.position.set(2, 3, 2); scene.add(sun);

// Face meshes (ABC and ACD, the two path faces)
function makeFaceMesh(color, opacity=0.22) {
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(9), 3));
  geo.setIndex([0,1,2]);
  geo.computeVertexNormals();
  return new THREE.Mesh(geo, new THREE.MeshStandardMaterial({
    color, side: THREE.DoubleSide, transparent: true, opacity,
    roughness: 0.4, depthWrite: false
  }));
}

const faceABC = makeFaceMesh(0x3d8ef0, 0.28); scene.add(faceABC);
const faceACD = makeFaceMesh(0x9b6bff, 0.28); scene.add(faceACD);
const faceABD = makeFaceMesh(0x334466, 0.15); scene.add(faceABD);
const faceBCD = makeFaceMesh(0x334466, 0.15); scene.add(faceBCD);

function setFace(mesh, p0, p1, p2) {
  const arr = mesh.geometry.attributes.position.array;
  arr[0]=p0.x;arr[1]=p0.y;arr[2]=p0.z;
  arr[3]=p1.x;arr[4]=p1.y;arr[5]=p1.z;
  arr[6]=p2.x;arr[7]=p2.y;arr[8]=p2.z;
  mesh.geometry.attributes.position.needsUpdate = true;
  mesh.geometry.computeVertexNormals();
}

// Edges
function makeLine(col, width=1) {
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(6), 3));
  const line = new THREE.Line(geo, new THREE.LineBasicMaterial({ color: col, linewidth: width }));
  scene.add(line); return line;
}
function setLine(line, p1, p2) {
  const arr = line.geometry.attributes.position.array;
  arr[0]=p1.x;arr[1]=p1.y;arr[2]=p1.z;
  arr[3]=p2.x;arr[4]=p2.y;arr[5]=p2.z;
  line.geometry.attributes.position.needsUpdate = true;
}

// All 6 edges
const eAB = makeLine(0xffcc44, 2); // opposite edges highlighted
const eCD = makeLine(0xff6644, 2);
const eAC = makeLine(0xaaaaaa); // crease
const eAD = makeLine(0x334488);
const eBC = makeLine(0x334488);
const eBD = makeLine(0x334488);

// Path line (M1 to M2): needs 20 segments for the bent 3D path vs straight unfolded
const PATH_N = 40;
const pathGeo = new THREE.BufferGeometry();
const pathArr = new Float32Array((PATH_N+1)*3);
pathGeo.setAttribute('position', new THREE.BufferAttribute(pathArr, 3));
const pathIndices = [];
for(let i=0;i<PATH_N;i++) pathIndices.push(i,i+1);
pathGeo.setIndex(pathIndices);

// Vertex dots
const dotMat = (col) => new THREE.MeshStandardMaterial({ color: col, roughness: 0.3 });
function dot(col, r=0.025) {
  const m = new THREE.Mesh(new THREE.SphereGeometry(r,12,8), dotMat(col));
  scene.add(m); return m;
}
const dA=dot(0xffffff), dB=dot(0xffffff), dC=dot(0xffffff), dD=dot(0xaaaaff);
const dM1=dot(0xffcc44, 0.04), dM2=dot(0xff6644, 0.04);

// Crease midpoint (midpoint of AC, where the path crosses in the unfolding)
const dCrease=dot(0xaaaaaa, 0.025);

let t = 0;

function update(frac) {
  t = frac;
  // Interpolated positions
  const pA = lerp(A3, A2, t), pB = lerp(B3, B2, t);
  const pC = lerp(C3, C2, t), pD = lerp(D3, Dp2, t);

  // Face geometry
  setFace(faceABC, pA, pB, pC);
  setFace(faceACD, pA, pC, pD);
  // The "other two" faces (ABD and BCD) only exist in 3D; fade them out
  const otherOpac = 0.15 * (1 - t);
  faceABD.material.opacity = otherOpac;
  faceBCD.material.opacity = otherOpac;
  if (t < 0.99) {
    setFace(faceABD, pA, pB, lerp(D3, D3, t)); // D stays at D3 for these other faces
    // Actually: ABD and BCD use the REAL D3 position, not D'
    const pD3real = lerp(D3, D3, 0); // always D3
    setFace(faceABD, pA, pB, D3);
    setFace(faceBCD, pB, pC, D3);
  }

  // Edges
  setLine(eAB, pA, pB);
  setLine(eCD, pC, pD);
  setLine(eAC, pA, pC);
  setLine(eAD, pA, D3);
  setLine(eBC, pB, pC);
  setLine(eBD, pB, D3);

  // M1, M2
  const M1 = pA.clone().lerp(pB, 0.5);
  const M2 = pC.clone().lerp(pD, 0.5);
  dM1.position.copy(M1);
  dM2.position.copy(M2);

  // Crease point (midpoint of AC)
  const crease = pA.clone().lerp(pC, 0.5);
  dCrease.position.copy(crease);

  // Path: in 3D it bends at the crease point; unfolded it's a straight line
  // 3D: M1→crease on face ABC, crease→M2 on face ACD
  // Unfolded: straight M1→M2
  const M1_3d = A3.clone().lerp(B3, 0.5);
  const crease_3d = A3.clone().lerp(C3, 0.5);
  const M2_3d = C3.clone().lerp(D3, 0.5);

  const M1_2d = A2.clone().lerp(B2, 0.5);
  const crease_2d = A2.clone().lerp(C2, 0.5);
  const M2_2d = C2.clone().lerp(Dp2, 0.5);

  for (let i = 0; i <= PATH_N; i++) {
    const s = i / PATH_N;
    // Interpolate path in 3D vs 2D
    let p3, p2;
    if (s <= 0.5) {
      const u = s * 2;
      p3 = M1_3d.clone().lerp(crease_3d, u);
      p2 = M1_2d.clone().lerp(M2_2d, s); // straight in 2D
    } else {
      const u = (s - 0.5) * 2;
      p3 = crease_3d.clone().lerp(M2_3d, u);
      p2 = M1_2d.clone().lerp(M2_2d, s);
    }
    const px = p3.x + t*(p2.x-p3.x);
    const py = p3.y + t*(p2.y-p3.y);
    const pz = p3.z + t*(p2.z-p3.z);
    pathArr[i*3]=px; pathArr[i*3+1]=py; pathArr[i*3+2]=pz;
  }
  pathGeo.attributes.position.needsUpdate = true;

  // Vertex dots
  dA.position.copy(pA); dB.position.copy(pB);
  dC.position.copy(pC); dD.position.copy(pD);
}
